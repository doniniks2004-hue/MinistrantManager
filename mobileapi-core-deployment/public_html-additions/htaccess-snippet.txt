# MobileAPI native routes. Root-level proxies intentionally avoid legacy
# catch-all rules such as ^api/(.*)$ -> api/$1.php present in older parishes.
RewriteRule ^api/v1/mobile/bootstrap/?$ mm-mobile-bootstrap.php [END,QSA]
RewriteRule ^api/v1/mobile/session/login/?$ mm-mobile-session-login.php [END,QSA]
RewriteRule ^api/v1/mobile/session/change-password/?$ mm-mobile-session-change-password.php [END,QSA]
RewriteRule ^api/v1/mobile/config/?$ mm-mobile-config.php [END,QSA]
RewriteRule ^api/v1/mobile/webview/handoff/?$ mm-mobile-webview-handoff.php [END,QSA]

# Production hardening for old maintenance scripts.
RewriteRule ^(?:install|reset|upd|fix|receiver)\.php$ - [F,END]
